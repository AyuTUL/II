#include<iostream>
using namespace std;
class Base
{
	public:
		virtual void show()
		{
			cout<<"Base show"<<endl;
		}
		void display()
		{
			cout<<"Base display"<<endl;
		}
};
class Derived: public Base
{
	public:
		virtual void show()
		{
			cout<<"Derived show"<<endl;
		}
		void display()
		{
			cout<<"Derived display"<<endl;
		}	
};
int main()
{
	Base b,*p;
	Derived d;
	cout<<"Base pointer pointing to base object :"<<endl;
	p=&b;
	p->show();
	p->display();
	cout<<"Base pointer pointing to derived object :"<<endl;
	p=&d;
	p->show();
	p->display();
	return 0;
}
